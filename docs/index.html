// << returns, xu spawns a coroutine, chan moves the value
@main
#errors

fn worker(task string, ch chan int) {
    ch <- task.len * 42
}

fn main() int {
    ch := chan int{cap: 2}
    xu worker("latency", ch)
    println(<-ch)
    << 0
}
