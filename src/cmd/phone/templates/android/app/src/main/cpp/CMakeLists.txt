cmake_minimum_required(VERSION 3.22)
project(xuphone LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native")
set_source_files_properties("${NATIVE}/app.o" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_library(xuphone SHARED bridge.cc "${NATIVE}/host/phone/runtime.cc" "${NATIVE}/host/phone/host.cc" "${NATIVE}/host/xu_ui_mem.cc" "${NATIVE}/app.o")
target_include_directories(xuphone PRIVATE "${NATIVE}/host")
target_link_options(xuphone PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(xuphone PRIVATE android log)
