cmake_minimum_required(VERSION 3.22)
project(XuPhoneGame LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
if(ANDROID)
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
else()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(native/SDL EXCLUDE_FROM_ALL)
if(ANDROID)
  target_link_options(SDL3-shared PRIVATE "-Wl,-z,max-page-size=16384")
endif()
file(GLOB NATIVE_ARCHIVES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/app.xuobj.native/native-*.a")
set(APP_OBJECT "${CMAKE_CURRENT_SOURCE_DIR}/native/app.o")
set_source_files_properties("${APP_OBJECT}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
set(GAME_SOURCES App.c native/host/phone/runtime.cc native/host/phone/game_runtime.cc
  native/host/phone/game_context.S "${APP_OBJECT}")
if(NATIVE_ARCHIVES)
  list(APPEND GAME_SOURCES native/app.xuobj.native/bridge.c)
endif()
if(ANDROID)
  add_library(main SHARED ${GAME_SOURCES})
  set(GAME_TARGET main)
  target_link_options(main PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined")
else()
  add_executable(XuPhoneGame ${GAME_SOURCES})
  set_target_properties(XuPhoneGame PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist")
  set(GAME_TARGET XuPhoneGame)
  target_link_options(XuPhoneGame PRIVATE "-Wl,-export_dynamic")
  add_custom_command(TARGET XuPhoneGame POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/resources"
      "$<TARGET_BUNDLE_DIR:XuPhoneGame>/xu-resources")
endif()
target_compile_definitions(${GAME_TARGET} PRIVATE XU_PHONE_GAME=1)
target_link_libraries(${GAME_TARGET} PRIVATE ${NATIVE_ARCHIVES} SDL3::SDL3)
if(ANDROID)
  target_link_libraries(main PRIVATE log)
endif()
